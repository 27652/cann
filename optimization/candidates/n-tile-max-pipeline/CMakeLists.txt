cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(n_tile_max LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 14)
add_executable(bench_n_tile bench.asc)
target_compile_options(bench_n_tile PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
target_link_libraries(bench_n_tile PRIVATE tiling_api register platform unified_dlog dl m graph_base)
add_executable(bench_n_tile_probe bench.asc)
target_compile_definitions(bench_n_tile_probe PRIVATE N_TILE_MIN_ROWS=17U)
target_compile_options(bench_n_tile_probe PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
target_link_libraries(bench_n_tile_probe PRIVATE tiling_api register platform unified_dlog dl m graph_base)
target_compile_definitions(bench_n_tile_probe PRIVATE N_TILE_ONE_CORE=1)
