cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(architecture_trials LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 14)
foreach(variant local-k local-k-safe scale-owner tile-fifo)
    add_executable(bench_${variant} bench.asc)
    target_compile_definitions(bench_${variant} PRIVATE BENCH_KERNEL="${CMAKE_CURRENT_SOURCE_DIR}/${variant}/kernel.asc")
    target_compile_options(bench_${variant} PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
    target_link_libraries(bench_${variant} PRIVATE tiling_api register platform unified_dlog dl m graph_base)
endforeach()
