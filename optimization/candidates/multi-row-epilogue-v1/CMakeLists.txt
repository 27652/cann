cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(multi_row_epilogue LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 14)
add_executable(quant_matmul_relu_quant_custom main.asc)
foreach(rows 1 2 4 8)
    add_executable(bench_rows${rows} bench.asc)
    target_compile_definitions(bench_rows${rows} PRIVATE QUANT_MULTI_ROW=${rows}U)
endforeach()
foreach(target quant_matmul_relu_quant_custom bench_rows1 bench_rows2 bench_rows4 bench_rows8)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
    target_link_libraries(${target} PRIVATE tiling_api register platform unified_dlog dl m graph_base)
endforeach()
