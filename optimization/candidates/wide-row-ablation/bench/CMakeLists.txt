cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(wide_row_ablation LANGUAGES ASC CXX)
set(CANDIDATE "baseline" CACHE STRING "Isolated ablation variant")
get_filename_component(HERE "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
add_executable(bench_ablation "${HERE}/../wide-row-tile-cv/bench.asc")
target_compile_options(bench_ablation PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
target_compile_definitions(bench_ablation PRIVATE BENCH_KERNEL="${HERE}/${CANDIDATE}/kernel.asc")
target_link_libraries(bench_ablation PRIVATE tiling_api register platform unified_dlog dl m graph_base)
set(SIM_LIB "$ENV{ASCEND_HOME_PATH}/x86_64-linux/simulator/dav_2201/lib")
if(EXISTS "${SIM_LIB}/libascendcl.so")
    target_link_libraries(bench_ablation PRIVATE "${SIM_LIB}/libascendcl.so")
    set_target_properties(bench_ablation PROPERTIES BUILD_RPATH "${SIM_LIB}")
endif()
