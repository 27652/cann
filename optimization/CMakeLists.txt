cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(quant_bench LANGUAGES ASC CXX)
foreach(variant baseline optimized prior v2 v3 vector cache tile128 mask)
    add_executable(bench_${variant} bench.asc)
    target_compile_options(bench_${variant} PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
    target_link_libraries(bench_${variant} PRIVATE tiling_api register platform unified_dlog dl m graph_base)
endforeach()
target_compile_definitions(bench_baseline PRIVATE BENCH_KERNEL="baseline/kernel.asc")
target_compile_definitions(bench_prior PRIVATE BENCH_KERNEL="../op/kernel.asc")
target_compile_definitions(bench_v2 PRIVATE BENCH_KERNEL="../op/kernel_v2.asc")
target_compile_definitions(bench_v3 PRIVATE BENCH_KERNEL="../op/kernel_v3.asc")
target_compile_definitions(bench_vector PRIVATE QUANT_VECTOR_ONLY=1 QUANT_MASK_FAST_PATH=0)
target_compile_definitions(bench_cache PRIVATE QUANT_CACHE_TWO_TILES=1)
target_compile_definitions(bench_tile128 PRIVATE QUANT_BASE_N=128U)
target_compile_definitions(bench_mask PRIVATE QUANT_MASK_FAST_PATH=1)
